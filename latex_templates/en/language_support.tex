% English localization support
% Keep English captions and use digits in fncychap (Bjarne style) chapter headings.

\AtBeginDocument{
  % --- 1. Fix fncychap (Bjarne style) numbering ---
  % Standard Bjarne uses spelled-out English words (ONE, TWO...).
  % We override \DOCH to use digits (\thechapter).
  \makeatletter
  \@ifpackageloaded{fncychap}{
    \renewcommand{\DOCH}{%
      \mghrulefill{\RW}\par\nobreak
      \FmN{\chaptername}\space \thechapter\par\nobreak
      \mghrulefill{\RW}\par\nobreak
      \vskip 40\p@
    }
    \renewcommand{\DOTIS}[1]{%
      \raggedright
      \CTV\FmTi{#1}\par\nobreak
      \vskip 40\p@
    }
  }{}
  \makeatother

  % --- 2. English captions ---
  \gdef\contentsname{Contents}
  \gdef\chaptername{Chapter}
  \gdef\partname{Part}
  \gdef\appendixname{Appendix}
  \gdef\figurename{Figure}
  \gdef\tablename{Table}
  \gdef\bibname{Bibliography}
  \gdef\indexname{Index}
}
